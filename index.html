Streamlit App Script (app.py)

Use this script to run your saved sleep_disorder_rf_model.pkl and feature_scaler.pkl files locally.

import streamlit as st
import pandas as pd
import joblib

# load saved model and scaler
rf_model = joblib.load('sleep_disorder_rf_model.pkl')
scaler = joblib.load('feature_scaler.pkl')

st.title("Sleep Disorder Prediction System")

# collect user inputs
gender = st.selectbox("Gender", ["Male", "Female"])
age = st.number_input("Age", min_value=18, max_value=90, value=35)
sleep_duration = st.number_input("Sleep Duration (hours)", min_value=3.0, max_value=12.0, value=7.2, step=0.1)
quality_of_sleep = st.number_input("Quality of Sleep (1 to 10)", min_value=1, max_value=10, value=7)
physical_activity = st.number_input("Physical Activity Level (mins/day)", min_value=0, max_value=240, value=60)
stress_level = st.number_input("Stress Level (1 to 10)", min_value=1, max_value=10, value=5)
bmi_category = st.selectbox("BMI Category", ["Normal", "Overweight", "Obese"])
systolic_bp = st.number_input("Systolic Blood Pressure", min_value=90, max_value=200, value=125)
diastolic_bp = st.number_input("Diastolic Blood Pressure", min_value=50, max_value=130, value=80)
heart_rate = st.number_input("Heart Rate (bpm)", min_value=45, max_value=140, value=70)
daily_steps = st.number_input("Daily Steps", min_value=500, max_value=25000, value=7000)

if st.button("Predict Sleep Disorder"):
    # encode categorical inputs
    gender_val = 0 if gender == "Male" else 1
    bmi_map = {"Normal": 0, "Overweight": 1, "Obese": 2}
    bmi_val = bmi_map[bmi_category]

    # create dataframe matching training feature order
    feature_cols = [
        'Gender', 'Age', 'Sleep Duration', 'Quality of Sleep',
        'Physical Activity Level', 'Stress Level', 'BMI Category',
        'Systolic_BP', 'Diastolic_BP', 'Heart Rate', 'Daily Steps'
    ]

    input_df = pd.DataFrame([[
        gender_val, age, sleep_duration, quality_of_sleep,
        physical_activity, stress_level, bmi_val,
        systolic_bp, diastolic_bp, heart_rate, daily_steps
    ]], columns=feature_cols)

    # scale continuous columns
    continuous_cols = [
        'Age', 'Sleep Duration', 'Physical Activity Level',
        'Stress Level', 'Systolic_BP', 'Diastolic_BP',
        'Heart Rate', 'Daily Steps'
    ]
    input_df[continuous_cols] = scaler.transform(input_df[continuous_cols])

    # predict disorder class
    pred_class = rf_model.predict(input_df)[0]
    pred_probs = rf_model.predict_proba(input_df)[0]

    labels = {0: "None", 1: "Insomnia", 2: "Sleep Apnea"}
    st.subheader(f"Predicted Disorder: {labels[pred_class]}")

    prob_df = pd.DataFrame({
        "Disorder": ["None", "Insomnia", "Sleep Apnea"],
        "Probability": [f"{p * 100:.1f}%" for p in pred_probs]
    })
    st.table(prob_df)
